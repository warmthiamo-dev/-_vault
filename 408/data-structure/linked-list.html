void SortData(LNode *L) {
    LNode *p, *q;
    int temp;
    for (p=L->next; p!=NULL; p=p->next)
        for (q=p->next; q!=NULL; q=q->next)
            if (p->data > q->data) {
                temp=p->data;
                p->data=q->data;
                q->data=temp;
            }
}

时间 O(n²)，辅助空间 O(1)。

第一轮：pre 指向 2，min 指向 1。执行 pre->next=min->next 后，2 的后继变成 NULL；tail->next=min 将 1 接入新链表，随后 tail=min。

void SortLink(LNode *L) {
    LNode *head=L->next, *tail=L;
    L->next=NULL;
    while (head!=NULL) {
        LNode *min=head, *pre=NULL;
        for (LNode *p=head; p->next!=NULL; p=p->next)
            if (p->next->data < min->data) {
                min=p->next;
                pre=p;
            }
        if (pre==NULL) head=min->next;
        else pre->next=min->next;
        tail->next=min;
        tail=min;
        tail->next=NULL;
    }
}

时间 O(n²)，辅助空间 O(1)。

以处理 3 为例，q=p->next 保存 2；p->next=L->next 让 3 指向 4；L->next=p 让头结点指向 3；p=q 移至 2。

void Reverse(LNode *L) {
    LNode *p=L->next, *q;
    L->next=NULL;
    while (p!=NULL) {
        q=p->next;        // 保存后继
        p->next=L->next;  // 头插
        L->next=p;
        p=q;             // 移动
    }
}

时间 O(n)，辅助空间 O(1)。记忆：保存 → 头插 → 移动。

最终代码（保留用户的初始化）

void Delete(LNode *head, int n) {
    int *A=(int *)calloc(n+1,sizeof(int));
    if (A==NULL) return; // 分配失败保护
    LNode *p=head->next, *q=head->next;
    while (p!=NULL) {
        A[abs(p->data)]++;
        if (A[abs(p->data)]>1) {
            q->next=p->next;
            free(p);
            p=q->next;  // q 不移动
        } else {
            q=p;
            p=p->next;
        }
    }
    free(A);
}

通用删除模板应令 q=head、p=head->next，才能删除第一个数据结点；本题首结点必保留，故用户的双指针同起点写法成立。计数数组长度为 n+1。无需内层 while，避免跳过结点时漏计数。

总时间 O(m+n)（含初始化），辅助空间 O(n)。

04 · 快慢指针查找中间结点

最终采用用户更容易理解的拆步写法：快指针 q 的两次移动分开；第一次移动后，若 q 为空则奇数结束，不移动 p；若 q 为尾结点则偶数结束，p 再走一步取右中点。

LNode *FindMid(LNode *L) {
    LNode *p=L->next, *q=L->next;
    while (q!=NULL) {
        q=q->next;
        if (q==NULL) break;       // 奇数：p 不动
        if (q->next==NULL) {
            p=p->next;           // 偶数：右中点
            break;
        }
        q=q->next;
        p=p->next;
    }
    return p;                    // mid=p，不是 q
}

最终完整 C 代码（使用拆步中点函数）

// FindMid 使用上一节最终版本
void Reorder(LNode *L) {
    LNode *mid=FindMid(L);
    if (mid==NULL) return;
    LNode *p=L->next, *q, *temp;
    while (mid->next!=NULL) {
        q=mid;
        while (q->next->next!=NULL)
            q=q->next;           // 尾结点前驱
        temp=q->next;
        q->next=NULL;            // 摘下尾结点
        temp->next=p->next;
        p->next=temp;            // 后插
        p=p->next->next;         // 跨过新插入结点
    }
}

由于每次都重新寻找尾结点前驱，总时间 O(n²)，辅助空间 O(1)。该暴力版满足空间要求，时间非最优，可能损失效率分。本节不引入其他优化算法。
